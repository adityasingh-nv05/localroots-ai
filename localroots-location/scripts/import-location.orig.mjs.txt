import { config } from '../docs/config.mjs';
import { createPgClient } from './utils/pg-client.mjs';
import { validateDistrictRow } from './utils/validator.mjs';
import { promises as fs } from 'fs';
import { createReadStream } from 'fs';
import { pipeline } from 'stream/promises';
import crypto from 'crypto';
import { v4 as uuidv4 } from 'uuid';

/**
 * Minimal skeleton for the location‑data import pipeline.
 * – Loads environment variables (POSTGRES_URL, POSTGRES_PASSWORD).
 * – Generates a UUID `importRunId` for this execution.
 * – For each CSV file it:
 *   1. Starts a DB transaction.
 *   2. Creates (or re‑uses) a persistent staging table scoped by `import_run_id`.
 *   3. Streams the CSV into the staging table via PostgreSQL `COPY`.
 *   4. Runs simple validation helpers (example: LGD code format).
 *   5. Performs an idempotent upsert into the target table.
 *   6. Commits the transaction and writes a JSON‑line to a local manifest.
 *
 * The actual business logic (batch sizing, error handling, logging) will be added later.
 */

async function main() {
  const pg = await createPgClient();
  const importRunId = uuidv4();

  // Example processing for districts – repeat for other entities.
  const districtsFile = config.paths.districtsCsv;
  const checksum = await checksumFile(districtsFile);

  const client = await pg.connect();
  try {
    await client.query('BEGIN');
    // Create (or ensure) persistent staging table.
    await client.query(`
      CREATE TABLE IF NOT EXISTS stg_districts (
        import_run_id UUID NOT NULL,
        lgd_code TEXT NOT NULL,
        name TEXT NOT NULL,
        state_lgd_code TEXT NOT NULL,
        PRIMARY KEY (import_run_id, lgd_code)
      );
    `);

    // Stream CSV into staging table using COPY FROM STDIN.
    const copyQuery = `COPY stg_districts (import_run_id, lgd_code, name, state_lgd_code) FROM STDIN WITH (FORMAT csv, HEADER true)`;
    const stream = client.query(copyQuery);
    const fileStream = createReadStream(districtsFile);
    // Prepend the importRunId column via a transform – for brevity we assume the CSV already contains it.
    await pipeline(fileStream, stream);

    // Simple validation – placeholder.
    const { rows: toValidate } = await client.query('SELECT * FROM stg_districts WHERE import_run_id = $1', [importRunId]);
    for (const row of toValidate) {
      if (!validateDistrictRow(row)) {
        throw new Error(`Invalid district row: ${JSON.stringify(row)}`);
      }
    }

    // Upsert into real table.
    await client.query(`
      INSERT INTO public.districts (lgd_code, name, state_ut_id)
      SELECT d.lgd_code, d.name, s.id
      FROM stg_districts d
      JOIN public.states_uts s ON s.lgd_code = d.state_lgd_code
      ON CONFLICT (lgd_code) DO UPDATE SET name = EXCLUDED.name;
    `);

    await client.query('COMMIT');

    // Write manifest entry after successful commit.
    const manifestLine = {
      importRunId,
      entity: 'districts',
      sourceChecksum: checksum,
      rowCount: toValidate.length,
      timestamp: new Date().toISOString()
    };
    await fs.appendFile(config.paths.manifest, JSON.stringify(manifestLine) + '\n');
  } catch (err) {
    await client.query('ROLLBACK');
    console.error('Import batch failed – transaction rolled back', err);
    process.exit(1);
  } finally {
    client.release();
    await pg.end();
  }
}

async function checksumFile(path: string): Promise<string> {
  const hash = crypto.createHash('sha256');
  const file = createReadStream(path);
  for await (const chunk of file) {
    hash.update(chunk);
  }
  return hash.digest('hex');
}

main().catch(console.error);
